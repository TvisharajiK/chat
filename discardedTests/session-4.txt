func TestPostMessage_Success_200(t *testing.T) {
	// Arrange
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		assert.Equal(t, http.MethodPost, r.Method)
		assert.Equal(t, "Bearer testToken", r.Header.Get("Authorization"))
		assert.Equal(t, "application/json; charset=utf-8", r.Header.Get("Content-Type"))
		assert.Equal(t, "gzip", r.Header.Get("Content-Encoding"))
		assert.Equal(t, "gzip", r.Header.Get("Accept-Encoding"))

		// Check request body (decompress gzip)
		gzr, err := gzip.NewReader(r.Body)
		require.NoError(t, err)
		defer gzr.Close()
		bodyBytes, err := io.ReadAll(gzr)
		require.NoError(t, err)
		var reqBody map[string]string
		err = json.Unmarshal(bodyBytes, &reqBody)
		require.NoError(t, err)
		assert.Equal(t, "value", reqBody["key"])

		// Send gzipped response
		w.Header().Set("Content-Type", "application/json")
		w.Header().Set("Content-Encoding", "gzip")
		w.WriteHeader(http.StatusOK)
		gw := gzip.NewWriter(w)
		defer gw.Close()
		json.NewEncoder(gw).Encode(&batchResponse{
			SuccessCount: 1,
			FailureCount: 0,
			Responses:    []*tnpgResponse{{MessageID: "msg1"}},
		})
	}))
	defer server.Close()
	http.DefaultClient = server.Client() // Use test client
	t.Cleanup(func() { http.DefaultClient = defaultHTTPClient }) // Restore default client


	body := map[string]string{"key": "value"}
	config := &configType{AuthToken: "testToken"}

	// Act
	resp, err := postMessage(server.URL, body, config)

	// Assert
	require.NoError(t, err)
	require.NotNil(t, resp)
	assert.Equal(t, http.StatusOK, resp.httpCode)
	assert.Equal(t, "200 OK", resp.httpStatus)
	assert.Equal(t, 1, resp.SuccessCount)
	assert.Equal(t, 0, resp.FailureCount)
	assert.Empty(t, resp.FatalCode)
	assert.Empty(t, resp.FatalMessage)
	require.Len(t, resp.Responses, 1)
	assert.Equal(t, "msg1", resp.Responses[0].MessageID)
}
